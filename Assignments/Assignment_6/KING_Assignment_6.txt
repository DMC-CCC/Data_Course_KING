dat_means <- dat_sample_type %>%
  group_by(Substrate, Dilution, `Sample ID`, Time_Hours) %>% 
  summarise(Mean_absorbance = mean(Absorbance, na.rm = TRUE), .groups = "drop")
#First we need to calculate the mean absorbance of things.

dat_Itaconic_Acid <- dat_means %>%
  filter(Substrate == "Itaconic Acid")
#This creates the tibble to graph the mean absorbance over time for Itaconic Acid. Can be modified for any substrate in the data. 

anim_plot <- ggplot(data = dat_Itaconic_Acid, 
                    mapping = aes(x = Time_Hours, 
                                  y = Mean_absorbance, 
                                  color = `Sample ID`,        
                                  group = `Sample ID`)) +     
  geom_line(size = 0.8) +                  
  geom_point(size = 1.5) +                 
  facet_wrap(~ Dilution) +              
  theme_minimal() + 
  scale_x_continuous(
    breaks = c(50, 100, 150), 
    limits = c(0, 150)
  ) + 
  labs(
    title = "Absorbance Progression for Itaconic Acid",
    x = "Time",                    
    y = "Mean_absorbance",          
    color = "Sample ID"             
  ) +
  transition_reveal(Time_Hours)
  
animate(anim_plot, renderer = gifski_renderer(), width = 800, height = 450, res = 100)
